# ── File attachments: text extraction (optional but recommended) ─
# Without these, PDFs/DOCX are still uploaded to tools — only inline
# text extraction is skipped. Plain text/code/csv/json need nothing.
pypdf
python-docx

# ── /boq: real quantity takeoff from a CAD drawing (optional) ────
# Reads DXF geometry directly — no AI ever measures the drawing, so
# the numbers are real, not guessed. A .dwg file still needs local
# conversion to .dxf first; see core/boq.py's module docstring.
ezdxf

# ── Voice input (SPACE-to-speak at the prism › prompt) — OPT-IN ─
# Commented out so the main installer never breaks: pyaudio needs
# PortAudio on the system first (macOS: `brew install portaudio`,
# Debian/Ubuntu: `sudo apt install portaudio19-dev`). Then:
#   pip install pyaudio
# Without it, the prompt is typed-only (exactly as before).
# pyaudio
#
# The standalone whisper.py reference app additionally needs:
#   pip install numpy aiohttp

# ── /gerber: PCB fabrication data from Gerber + drill files (optional) ──
# Board size, minimum track width/spacing, drill sizes and hole counts are
# MEASURED from the real geometry — no AI ever sees a Gerber file, so the
# customer's design never leaves the machine. shapely does the copper
# geometry; without it everything except track SPACING still works.
shapely

# Prism Studio films an HTML page frame by frame — this is the browser engine.
playwright>=1.44

# /step reads the customer's 3D CAD model (sheet metal or plastic
# moulding) with a real B-rep kernel, offline — same rule as the
# Gerber add-on: the design never leaves the machine.
cadquery

# ── Reading the text in a picture a customer sends (core/ocr.py) ─────────
# A brochure crop with the product code printed under the photo is a
# common inquiry. RapidOCR carries its own models (~15 MB) and runs on the
# ONNX runtime (~15 MB wheel) -- the same on Windows, macOS and Linux, no
# system install, nothing sent anywhere. Optional at runtime (core.ocr
# falls back to the OS reader on a Mac and says what is missing elsewhere),
# bundled in every build so a Windows customer never meets that fallback.
rapidocr-onnxruntime
